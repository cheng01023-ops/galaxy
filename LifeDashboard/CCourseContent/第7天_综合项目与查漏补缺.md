### 步骤 2：添加学生

```c
void addStudent(void) {
    if (count >= 100) {
        printf("人数已满！\n");
        return;
    }
    printf("姓名 成绩：");
    scanf("%s %d", students[count].name, &students[count].score);
    count++;
    printf("已添加 %s，当前共 %d 人\n", students[count - 1].name, count);
}
```

### 步骤 3：查看学生

```c
void listStudents(void) {
    if (count == 0) {
        printf("（还没有学生）\n");
        return;
    }
    printf("\n%-4s %-16s %s\n", "序号", "姓名", "成绩");
    printf("------------------------\n");
    for (int i = 0; i < count; i++) {
        printf("%-4d %-16s %d\n", i + 1, students[i].name, students[i].score);
    }
}
```

### 步骤 4：按成绩排序（从高到低，冒泡排序）

```c
void sortByScore(void) {
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - 1 - i; j++) {
            if (students[j].score < students[j + 1].score) {
                Student t = students[j];
                students[j] = students[j + 1];
                students[j + 1] = t;
            }
        }
    }
    printf("排序完成！\n");
}
```

### 步骤 5：删除学生（按姓名）

```c
void deleteStudent(void) {
    char name[32];
    printf("要删除的学生姓名：");
    scanf("%s", name);

    int found = -1;
    for (int i = 0; i < count; i++) {
        if (strcmp(students[i].name, name) == 0) {
            found = i;
            break;
        }
    }
    if (found == -1) {
        printf("没有找到 %s\n", name);
        return;
    }

    // 把后面的学生往前挪一位，覆盖被删的
    for (int i = found; i < count - 1; i++) {
        students[i] = students[i + 1];
    }
    count--;
    printf("已删除 %s\n", name);
}
```

### 步骤 6：修改成绩（按姓名）

```c
void updateScore(void) {
    char name[32];
    printf("要修改的学生姓名：");
    scanf("%s", name);

    for (int i = 0; i < count; i++) {
        if (strcmp(students[i].name, name) == 0) {
            printf("%s 当前成绩 %d，新成绩：", name, students[i].score);
            scanf("%d", &students[i].score);
            printf("已更新\n");
            return;
        }
    }
    printf("没有找到 %s\n", name);
}
```

### 步骤 7：保存到文件 / 启动加载

```c
void saveToFile(void) {
    FILE *fp = fopen("students.txt", "w");
    if (fp == NULL) {
        printf("保存失败！\n");
        return;
    }
    for (int i = 0; i < count; i++) {
        fprintf(fp, "%s %d\n", students[i].name, students[i].score);
    }
    fclose(fp);
    printf("已保存 %d 名学生到 students.txt\n", count);
}

void loadFromFile(void) {
    FILE *fp = fopen("students.txt", "r");
    if (fp == NULL) return;          // 文件不存在就跳过，不报错
    while (count < 100 &&
           fscanf(fp, "%s %d", students[count].name, &students[count].score) == 2) {
        count++;
    }
    fclose(fp);
    printf("已从文件加载 %d 名学生\n", count);
}
```

### 步骤 8：组装 main，测试所有功能

```c
int main(void) {
    loadFromFile();

    while (1) {
        showMenu();
        int choice;
        scanf("%d", &choice);

        if (choice == 0) break;
        switch (choice) {
            case 1: addStudent();    break;
            case 2: listStudents();  break;
            case 3: sortByScore();   break;
            case 4: deleteStudent(); break;
            case 5: updateScore();   break;
            case 6: saveToFile();    break;
            default: printf("无效选项！\n");
        }
    }
    printf("再见！\n");
    return 0;
}
```

## 四、常见 Bug 自查清单（调试三板斧）

| 现象 | 可能原因 | 检查方法 |
|---|---|---|
| 程序崩溃 | 数组越界 / 野指针 | 检查下标是否在 0~n-1，指针是否初始化 |
| 输出乱码 | scanf 忘加 `&` / 字符串没 `\0` | 逐个检查 scanf，检查字符数组 |
| 死循环 | 循环变量没更新 / 条件恒真 | 在循环体加 printf 打印中间值 |
| 结果不对 | 整数除法丢小数 / `==` 写成 `=` | 打印中间结果，逐步排查 |
| 读文件失败 | 路径不对 / 忘判空 | 打印 fp 是否为 NULL |

**调试技巧：**

1. **printf 大法**：在关键位置打印变量的值，看它走到哪、值是多少
2. **二分定位**：程序一大段出错，先注释一半，确定错误在哪个函数
3. **Xcode 断点**：点击行号加断点，运行后在调试区查看所有变量（比 printf 更强）
4. **最小复现**：把问题缩小到最短能触发 bug 的代码，再分析

### 5.2 自测题（能答上来就是真会了）

1. `int x = 5; int *p = &x; *p = 10;` 之后 x 等于几？为什么？
2. `7 / 2` 和 `7.0 / 2` 分别等于什么？为什么不同？
3. 为什么 swap 函数必须传指针？
4. `char s[10] = "hi";` 里 `\0` 存在哪里？`strlen(s)` 返回几？
5. 链表和数组各自适合什么场景？
6. 写 `fopen` 之后第一件事应该做什么？为什么？

<details>
<summary>参考答案（先自己写，再对照）</summary>

1. x = 10。p 指向 x，`*p = 10` 等价于 `x = 10`（通过地址修改原变量）。
2. `7 / 2 = 3`（整数除法丢小数）；`7.0 / 2 = 3.5`（有一边是小数就是小数除法）。
3. 形参是实参的副本，传值只能改副本；传指针（地址）才能在函数里改原变量。
4. `\0` 存在数组下标 2 的位置（`s[2]`），`strlen` 返回 2（不含 \0）。
5. 数组适合大小固定、频繁按下标访问；链表适合频繁增删、大小动态变化。
6. 判断是否为 NULL。因为文件可能不存在或打不开，不判空直接使用会崩溃。
</details>
